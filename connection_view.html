async function fetchData() { try { // 1. सबसे पहले लिस्ट को खाली करें ताकि पुराना डेटा न दिखे masterRows = []; document.getElementById('tableBody').innerHTML = ""; stats = { total: 0, current: 0, arrear: 0, balance: 0 }; // 2. LocalStorage से डेटा लोड करें (यही वो डेटा है जो डिलीट करने पर हटना चाहिए) let localData = JSON.parse(localStorage.getItem("newConnections")) || []; // 3. CSV से डेटा लोड करें let csvRows = []; try { const response = await fetch('Connection_data.csv'); if(response.ok) { const csvData = await response.text(); const lines = csvData.split(/\r?\n/).filter(line => line.trim() !== "").slice(1); csvRows = lines.map(line => { const cols = line.split(/,(?=(?:(?:[^"]*"){2})*[^"]*$)/); if (cols.length >= 9) { return createRowObject(cols[0], cols[1], cols[2], cols[5], cols[6], cols[7], cols[8], false); } return null; }).filter(x => x); } } catch(e) { console.log("CSV load skipped"); } // 4. लोकल और CSV डेटा को मिलाएं // ध्यान दें: अब localRows को पहले रखा गया है ताकि ताजा डेटा ऊपर दिखे let localRows = localData.map(d => createRowObject(d.id, d.ward, d.name, d.mobile, d.curr, d.arr, d.total, true)); masterRows = [...localRows, ...csvRows]; // 5. स्टेट्स की गणना करें और टेबल दिखाएँ calculateStats(); renderTable(masterRows); } catch (e) { console.error("Error loading data:", e); } } // यह फंक्शन सुनिश्चित करता है कि नंबर सही से जुड़ें (₹010600 वाली समस्या को खत्म करता है) function createRowObject(id, ward, name, mobile, curr, arr, total, isNew) { const cVal = parseFloat(curr) || 0; const aVal = parseFloat(arr) || 0; const tVal = parseFloat(total) || 0; const item = { id: id.toString().startsWith("NPPNCN") ? id : `NPPNCN-${id.toString().trim()}`, ward: (ward || "").trim(), name: (name || "").replace(/"/g, '').trim(), mobile: (mobile || "---").trim(), curr: cVal, arr: aVal, total: tVal, rawId: id.toString().replace("NPPNCN-", "").trim() }; return { data: item, html: ` ${item.id}${isNew ? 'NEW' : ''} ${item.ward} ${item.name} ${item.mobile} ₹${item.curr.toFixed(2)} ₹${item.arr.toFixed(2)} ₹${item.total.toFixed(2)} `, text: `${item.id} ${item.ward} ${item.name} ${item.mobile}`.toUpperCase() }; }
