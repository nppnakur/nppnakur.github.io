function doGet(e) { var action = e.parameter.action; var id = e.parameter.id; var callback = e.parameter.callback; // यह लाइन बहुत जरूरी है var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = sheet.getDataRange().getValues(); var output = ""; if (action == 'read') { var foundData = []; for (var i = 1; i < data.length; i++) { if (data[i][0].toString().trim() == id.toString().trim()) { foundData = data[i]; break; } } output = JSON.stringify(foundData); } if (action == 'delete') { var status = "Not Found"; for (var i = 1; i < data.length; i++) { if (data[i][0].toString().trim() == id.toString().trim()) { sheet.deleteRow(i + 1); status = "Success"; break; } } output = JSON.stringify(status); } // यह हिस्सा नेटवर्क एरर को खत्म कर देगा if (callback) { return ContentService.createTextOutput(callback + '(' + output + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(output) .setMimeType(ContentService.MimeType.JSON); } }
