function handleDelete() { const pass = prompt("हटाने के लिए एडमिन पासवर्ड दर्ज करें:"); if (pass === "admin123") { // ID को पूरी तरह साफ करें (सिर्फ नंबर और जरूरी अक्षर रखें) const currentId = document.getElementById('disp_id').value.replace(/["']/g, '').trim(); if (confirm("क्या आप इस कनेक्शन को हटाना चाहते हैं?")) { let localData = JSON.parse(localStorage.getItem("newConnections")) || []; // मैचिंग के लिए सख्त फिल्टर (ID को NPPNCN- के साथ या बिना भी चेक करेगा) const initialLength = localData.length; localData = localData.filter(item => { let itemId = item.id.toString().replace(/["']/g, '').trim(); // अगर ID में NPPNCN- नहीं है तो जोड़कर चेक करें if (!itemId.startsWith("NPPNCN-")) itemId = "NPPNCN-" + itemId; return itemId !== currentId; }); if (localData.length < initialLength) { localStorage.setItem("newConnections", JSON.stringify(localData)); alert("सफलतापूर्वक हटा दिया गया!"); window.location.href = "connection_detail.html"; } else { alert("त्रुटि: यह डेटा LocalStorage में नहीं मिला। (हो सकता है यह CSV फाइल में हो जिसे ब्राउज़र से नहीं हटाया जा सकता)"); } } } else if (pass !== null) { alert("गलत पासवर्ड!"); } }
